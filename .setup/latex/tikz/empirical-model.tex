\begin{tikzpicture}[

		var/.style={draw, circle, minimum width=1.7cm, minimum height=0.9cm, align=center},
		parbox/.style={draw, rounded corners=2pt, minimum width=5.6cm, minimum height=1.35cm, align=left},

		% arrowCR/.style={-Latex, thick, dashed},
		arrowAR/.style={-{Stealth[length=3.2mm,width=2.2mm]}, thick},

		extract/.style={-Latex, densely dotted, very thick},
		predict/.style={-Latex, very thick},
		attractorArrow/.style={-Latex, thick, dash dot},
		attrnode/.style={draw=none, circle, minimum size=0.9cm, inner sep=0pt, align=center},

	\draw[arrowCR] (NA_t) -- node[pos=0.25,above,sloped,lab] {$\mathrm{CR}(\mathrm{NA}\!\to\!\mathrm{PA})$} (PA_tp1);

	\node[attrnode, right=.35cm of PA_tp1] (nuPA) {$\mathrm{PA(Intercept)}$};
