``` r
  data = data,
  observed = c("na", "pa"),
  id = "id",
  center = TRUE,
  ncores = parallel::detectCores()
```

## Stage 2: Multivariate Random-Effects Meta-Analysis of the Dynamic Parameter System

``` r

library(metaDyn)
stage2 <- MetaVARMx(
  object = stage1,
  random = TRUE,
  ncores = parallel::detectCores()
)
```
